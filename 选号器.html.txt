<script>
const CONFIG = {
  ssq: { mainName:'红球', mainMax:33, mainPick:6, subName:'蓝球', subMax:16, subPick:1 },
  dlt: { mainName:'前区', mainMax:35, mainPick:5, subName:'后区', subMax:12, subPick:2 }
};

let currentType = 'ssq';
let selMain = new Set();
let selSub = new Set();
let results = [];

const $ = id => document.getElementById(id);
const range = (s, e) => Array.from({ length: e - s + 1 }, (_, i) => s + i);
const pad = n => String(n).padStart(2, '0');

function shuffle(a) {
  a = [...a];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
function pick(max, n) {
  return shuffle(range(1, max)).slice(0, n).sort((a, b) => a - b);
}
function genOne() {
  const c = CONFIG[currentType];
  return { main: pick(c.mainMax, c.mainPick), sub: pick(c.subMax, c.subPick) };
}

let toastTimer;
function flash(msg) {
  const t = $('toast');
  t.textContent = msg;
  t.classList.add('show');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => t.classList.remove('show'), 1400);
}

function renderPools() {
  const c = CONFIG[currentType];
  $('mainTitle').textContent = `${c.mainName}（选 ${c.mainPick} 个）`;
  $('subTitle').textContent = `${c.subName}（选 ${c.subPick} 个）`;

  const mp = $('mainPool');
  mp.innerHTML = '';
  for (let n = 1; n <= c.mainMax; n++) {
    const b = document.createElement('button');
    b.className = 'ball main' + (selMain.has(n) ? ' on' : '');
    b.textContent = pad(n);
    b.onclick = () => toggle('main', n);
    mp.appendChild(b);
  }

  const sp = $('subPool');
  sp.innerHTML = '';
  for (let n = 1; n <= c.subMax; n++) {
    const b = document.createElement('button');
    b.className = 'ball sub' + (selSub.has(n) ? ' on' : '');
    b.textContent = pad(n);
    b.onclick = () => toggle('sub', n);
    sp.appendChild(b);
  }
  updateBar();
}

function toggle(which, n) {
  const c = CONFIG[currentType];
  const set = which === 'main' ? selMain : selSub;
  const max = which === 'main' ? c.mainPick : c.subPick;
  const name = which === 'main' ? c.mainName : c.subName;

  if (set.has(n)) set.delete(n);
  else {
    if (set.size >= max) { flash(`${name}最多选 ${max} 个`); return; }
    set.add(n);
  }
  renderPools();
}

function updateBar() {
  const c = CONFIG[currentType];
  $('selectInfo').textContent =
    `${c.mainName} ${selMain.size}/${c.mainPick} · ${c.subName} ${selSub.size}/${c.subPick}`;

  const full = selMain.size === c.mainPick && selSub.size === c.subPick;
  const useBtn = $('useManual');
  useBtn.disabled = !full;
  useBtn.classList.toggle('ready', full);

  const showBtn = $('showResult');
  showBtn.disabled = !results.length;
  showBtn.textContent = results.length ? `结果 ${results.length}` : '结果';
}

function renderSheet() {
  const c = CONFIG[currentType];
  const body = $('sheetBody');
  if (!results.length) {
    body.innerHTML = '<div class="empty">还没有选号</div>';
    return;
  }
  body.innerHTML = results.map((item, i) => {
    const mains = item.main.map(n => `<span class="mini main">${pad(n)}</span>`).join('');
    const subs = item.sub.map(n => `<span class="mini sub">${pad(n)}</span>`).join('');
    return `<div class="result-item">
      <div class="result-idx">第 ${i + 1} 注</div>
      <div class="result-balls">${mains}<span class="divider"></span>${subs}</div>
    </div>`;
  }).join('');
}

function openSheet() {
  renderSheet();
  $('sheet').classList.add('show');
  $('sheetMask').classList.add('show');
}
function closeSheet() {
  $('sheet').classList.remove('show');
  $('sheetMask').classList.remove('show');
}

document.querySelectorAll('.tab').forEach(btn => {
  btn.onclick = () => {
    if (currentType === btn.dataset.type) return;
    currentType = btn.dataset.type;
    document.querySelectorAll('.tab').forEach(x => x.classList.toggle('active', x === btn));
    selMain.clear();
    selSub.clear();
    results = [];
    closeSheet();
    renderPools();
  };
});

$('randomOne').onclick = () => {
  results = [genOne()];
  updateBar();
  openSheet();
};

$('randomFive').onclick = () => {
  results = Array.from({ length: 5 }, genOne);
  updateBar();
  openSheet();
};

$('useManual').onclick = () => {
  const c = CONFIG[currentType];
  if (selMain.size !== c.mainPick || selSub.size !== c.subPick) return;
  results = [{
    main: [...selMain].sort((a, b) => a - b),
    sub: [...selSub].sort((a, b) => a - b)
  }];
  updateBar();
  openSheet();
};

$('clearBtn').onclick = () => {
  selMain.clear();
  selSub.clear();
  results = [];
  renderPools();
  closeSheet();
  flash('已清空');
};

$('showResult').onclick = openSheet;
$('closeSheet').onclick = closeSheet;
$('sheetMask').onclick = closeSheet;

$('copyBtn').onclick = () => {
  if (!results.length) return;
  const c = CONFIG[currentType];
  const text = results.map((item, i) =>
    `第${i + 1}注 ${c.mainName}：${item.main.map(pad).join(' ')}  ${c.subName}：${item.sub.map(pad).join(' ')}`
  ).join('\n');
  copyText(text);
};

function copyText(text) {
  const fallback = () => {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.position = 'fixed';
    ta.style.opacity = '0';
    document.body.appendChild(ta);
    ta.select();
    try {
      document.execCommand('copy');
      flash('已复制');
    } catch (e) {
      flash('复制失败，请长按选择');
    }
    document.body.removeChild(ta);
  };

  if (navigator.clipboard && window.isSecureContext) {
    navigator.clipboard.writeText(text).then(() => flash('已复制'), fallback);
  } else {
    fallback();
  }
}

renderPools();
</script>
